/* ====================================== TOOLNEST - AUTHENTICATION SUPABASE JS V2 ====================================== */ // 1. SUPABASE PROJECT URL const SUPABASE_URL = "https://kcfbjixhkxpyntdmnqhx.supabase.co"; // 2. SUPABASE PUBLISHABLE KEY const SUPABASE_PUBLISHABLE_KEY = "sb_publishable_zwSvLdZsnR0QdAgOUZ4Avw_znzNv8sk"; // 3. CHECK SUPABASE LIBRARY if (!window.supabase) { throw new Error( "Supabase JS v2 is missing. Load the Supabase library before auth.js." ); } // 4. VALIDATE CONFIGURATION if ( !SUPABASE_URL || !SUPABASE_PUBLISHABLE_KEY || SUPABASE_PUBLISHABLE_KEY === "PASTE_YOUR_REAL_PUBLISHABLE_KEY_HERE" ) { throw new Error( "Please check your Supabase URL and Publishable Key." ); } // 5. INITIALIZE SUPABASE CLIENT const supabaseClient = window.supabase.createClient( SUPABASE_URL, SUPABASE_PUBLISHABLE_KEY ); // 6. SIGN UP async function signUp(email, password) { if (!email || !password) { throw new Error("Email and password are required."); } const { data, error } = await supabaseClient.auth.signUp({ email: email.trim(), password: password }); if (error) { throw error; } return data; } // 7. LOGIN async function logIn(email, password) { if (!email || !password) { throw new Error("Email and password are required."); } const { data, error } = await supabaseClient.auth.signInWithPassword({ email: email.trim(), password: password }); if (error) { throw error; } return data; } // 8. LOGOUT async function logOut() { const { error } = await supabaseClient.auth.signOut(); if (error) { throw error; } } // 9. GET CURRENT USER async function getCurrentUser() { const { data, error } = await supabaseClient.auth.getUser(); if (error) { throw error; } return data.user; } // 10. GET CURRENT SESSION async function getCurrentSession() { const { data, error } = await supabaseClient.auth.getSession(); if (error) { throw error; } return data.session; } // 11. LISTEN FOR AUTHENTICATION CHANGES supabaseClient.auth.onAuthStateChange( (event, session) => { window.dispatchEvent( new CustomEvent("toolnest-auth-change", { detail: { event, session } }) ); } ); // 12. MAKE FUNCTIONS AVAILABLE GLOBALLY window.toolnestAuth = { signUp, logIn, logOut, getCurrentUser, getCurrentSession }; // 13. READY MESSAGE console.log("ToolNest authentication ready.");
